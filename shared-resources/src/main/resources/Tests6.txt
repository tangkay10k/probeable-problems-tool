/* Return a freshly-allocated C-string holding the expected output for
 * a given n.  Caller must free().                                     */
static char *build_expected(int n)
{
    if (n <= 0) {
        /* For n == 0 or negative: no output at all. */
        char *out = malloc(1);
        if (!out) { perror("malloc"); exit(EXIT_FAILURE); }
        out[0] = '\0';
        return out;
    }

    /* Worst-case line is "CheeseCake\n" (11 chars).  Allocate generously. */
    size_t cap = (size_t)n * 16 + 1;
    char *out = malloc(cap);
    if (!out) { perror("malloc"); exit(EXIT_FAILURE); }

    size_t len = 0;
    for (int i = 1; i <= n; ++i) {
        const char *tok = NULL;
        char numbuf[32];

        if (i % 14 == 0) {
            tok = "CheeseCake";
        } else if (i % 2 == 0) {
            tok = "Cheese";
        } else if (i % 7 == 0) {
            tok = "NoCheese";
        } else {
            snprintf(numbuf, sizeof(numbuf), "%d", i);
            tok = numbuf;
        }

        size_t toklen = strlen(tok);
        /* Ensure capacity for token + '\n' + '\0'. */
        if (len + toklen + 2 > cap) {
            cap *= 2;
            out = realloc(out, cap);
            if (!out) { perror("realloc"); exit(EXIT_FAILURE); }
        }

        memcpy(out + len, tok, toklen);
        len += toklen;
        out[len++] = '\n';
    }
    out[len] = '\0';
    return out;
}

/* Capture the output of SpecialDividers(n) and return it as a freshly
 * allocated string.  Caller must free().                              */
static char *capture_output(int n)
{
    /* Flush any pending stdout. */
    fflush(stdout);

    /* Duplicate the current stdout file descriptor so we can restore it. */
    int saved_fd = dup(fileno(stdout));
    if (saved_fd == -1) { perror("dup"); exit(EXIT_FAILURE); }

    /* Open a temporary file to grab the output. */
    FILE *tmp = tmpfile();
    if (!tmp) { perror("tmpfile"); exit(EXIT_FAILURE); }

    /* Redirect stdout to the temporary file. */
    if (dup2(fileno(tmp), fileno(stdout)) == -1) { perror("dup2"); exit(EXIT_FAILURE); }

    /* Call the function under test. */
    SpecialDividers(n);

    /* Flush to make sure everything is written. */
    fflush(stdout);

    /* Determine the size of what was written. */
    fseek(tmp, 0, SEEK_END);
    long sz = ftell(tmp);
    if (sz < 0) { perror("ftell"); exit(EXIT_FAILURE); }
    rewind(tmp);

    /* Read the data into a buffer. */
    char *buf = malloc((size_t)sz + 1);
    if (!buf) { perror("malloc"); exit(EXIT_FAILURE); }
    if (fread(buf, 1, (size_t)sz, tmp) != (size_t)sz) {
        perror("fread"); exit(EXIT_FAILURE);
    }
    buf[sz] = '\0';

    /* Restore original stdout. */
    fflush(stdout);
    if (dup2(saved_fd, fileno(stdout)) == -1) { perror("dup2 restore"); exit(EXIT_FAILURE); }
    close(saved_fd);
    fclose(tmp);

    return buf;
}

static void assert_output_equals(int n)
{
    char *expected = build_expected(n);
    char *actual   = capture_output(n);

    if (strcmp(expected, actual) != 0) {
        fprintf(stderr,
                "\nTest failed for n = %d\n"
                "Expected output:\n"
                "----------------\n%s"
                "----------------\n"
                "Actual output:\n"
                "----------------\n%s"
                "----------------\n",
                n, expected, actual);
        free(expected);
        free(actual);
        exit(EXIT_FAILURE);
    }

    free(expected);
    free(actual);
}
