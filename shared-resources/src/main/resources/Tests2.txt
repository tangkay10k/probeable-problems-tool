// Helper to capture stdout into a buffer and restore it
typedef struct { int old_fd; int tmp_fd; } Capture;

Capture start_capture(const char *tmp_filename) {
    Capture c;
    // Duplicate stdout
    c.old_fd = dup(STDOUT_FILENO);
    // Open temp file
    c.tmp_fd = open(tmp_filename, O_RDWR | O_CREAT | O_TRUNC, 0600);
    assert(c.tmp_fd >= 0);
    // Redirect stdout to temp file
    dup2(c.tmp_fd, STDOUT_FILENO);
    // Disable stdout buffering
    setbuf(stdout, NULL);
    return c;
}

char *stop_capture(Capture c) {
    // Ensure all data is written
    fsync(c.tmp_fd);
    lseek(c.tmp_fd, 0, SEEK_SET);

    static char buffer[256];
    ssize_t n = read(c.tmp_fd, buffer, sizeof(buffer) - 1);
    assert(n >= 0);
    buffer[n] = '\0';

    // Restore stdout
    dup2(c.old_fd, STDOUT_FILENO);
    close(c.tmp_fd);
    close(c.old_fd);
    return buffer;
}

char *capture_output(int values[], int length) {
    const char *tmpfile = "test_out.txt";
    Capture c = start_capture(tmpfile);
    SmallestEven(values, length);
    return stop_capture(c);
}

void test_no_evens() {
    int values[] = {1, 3, 5};
    char *out = capture_output(values, 3);
    assert(strcmp(out, "No evens") == 0);
}

void test_single_even() {
    int values[] = {3, 4, 1};
    // smallest even is 4 at index 1
    char *out = capture_output(values, 3);
    assert(strcmp(out, "1") == 0);
}

void test_multiple_occurrences() {
    int values[] = {2, 5, 2, 8};
    // smallest even is 2 at indices 0 and 2, prints "2 0"
    char *out = capture_output(values, 4);
    assert(strcmp(out, "2 0") == 0);
}

void test_negative_evens() {
    int values[] = {-4, -2, 0, 3};
    // smallest even is -4 at index 0, prints "0"
    char *out = capture_output(values, 4);
    assert(strcmp(out, "0") == 0);
}

void test_even_at_end_and_start() {
    int values[] = {6, 7, 8, 6};

    char *out = capture_output(values, 4);
