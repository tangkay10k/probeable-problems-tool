int CountBetween(int *values, int n, int a, int b) {
    if (n > sizeof(values)) {
        n = sizeof(values);
    }

    int count = 0;
    for (int i = 0; i < n; i += 1) {
        if ((a < values[i] && values[i] < b) || (b < values[i] && values[i] < a )) {
            count++;
        }

    }
    return count;
}
