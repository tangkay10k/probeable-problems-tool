char FirstVowel(char *s) {
    char vowels[5] = "aeiou";
    char upper[5] = "AEIOU";

    for (int i = 0; i < 5; i++) {
        for (int j = 0; j < strlen(s); j++) {
            if (s[j] == vowels[i] || s[j] == upper[i]) {
                return vowels[i];
            }
        }
    }
    return '-';
}
