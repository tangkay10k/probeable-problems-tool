void SmallestEven(int values[], int length) {

    int *smallest = malloc(length * sizeof(int));
    int smolNum = 100000;

    for (int i = 0; i < length; i += 1) {
        if (values[i] % 2 == 0 && values[i] < smolNum) {
           smolNum = values[i];
        }
    }

    for (int i = 0; i < length; i += 1) {
        if (values[i] == smolNum) {
            smallest[i] = 1;
        }
    }

    for (int i = length-1; i >=0; i-= 1) {
        if (smallest[i] == 1) {
            printf("%d ", i);
        }
    }

    if (smolNum == 100000) {
        printf("%s", "No evens");
    }
    return;

}
